#!/usr/bin/python3

# Program name: calculate_area.py
# Program purpose: This program greets the user and calculates the area of a square, a rectangle and a circle.

# constant variable(s)
PI = 3.14

# Take user's name as input.
Name = input("Enter your name: ")
print()
# In this section, we are calculating areas
print(f"Hello, {Name}! Let's first calculate the area of a square.")
aSideOfSquare = float(input("Enter the side of the square: "))
s_area = aSideOfSquare * aSideOfSquare
print(f"Now let's calculate the area of a rectangle.")
rectangle_length = float(input("Enter the length of the square: "))
Rectangle_Width = float(input("Enter the width of the rectangle: "))
r_area = rectangle_length * Rectangle_Width
print(f"Finally, let's calculate the area of a circle.")
a_circle_radius = float(input("Enter the radius of the circle: "))
circle_area = a_circle_radius * a_circle_radius * PI
# In this section we are printing!!
print()
print(f"The area of the square with side {aSideOfSquare} is {s_area}.")
print(f"The area of the rectangle with length {rectangle_length} and width {Rectangle_Width} is {r_area}.")
print(f"The area of the circle with radius {a_circle_radius} is {circle_area}.")
